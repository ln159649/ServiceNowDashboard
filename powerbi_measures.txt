POWER BI MEASURES (DAX) - ServiceNow incident dashboard
=========================================================
How to add each one: select the table named in the header, then Table tools > New measure,
paste the formula, press Enter. Set the format afterward (Measure tools > Format):
  - names ending in % : Percentage, 1 decimal
  - names ending in Hrs : Decimal number, 1 decimal
  - counts : Whole number, thousands separator

---------------------------------------------------------
TABLE: incident_facts
---------------------------------------------------------

Total Incidents = COUNTROWS(incident_facts)

Resolved Incidents =
CALCULATE(
    COUNTROWS(incident_facts),
    incident_facts[is_resolved] = TRUE()
)

Never Resolved = [Total Incidents] - [Resolved Incidents]

Median Resolution Hrs = MEDIAN(incident_facts[resolution_hours])

P90 Resolution Hrs = PERCENTILE.INC(incident_facts[resolution_hours], 0.9)

Resolved Within 24h % =
DIVIDE(
    CALCULATE(
        COUNTROWS(incident_facts),
        incident_facts[is_resolved] = TRUE(),
        incident_facts[resolution_hours] < 24
    ),
    [Resolved Incidents]
)

Over 7 Days % =
DIVIDE(
    CALCULATE(
        COUNTROWS(incident_facts),
        incident_facts[is_resolved] = TRUE(),
        incident_facts[resolution_hours] >= 168
    ),
    [Resolved Incidents]
)

SLA Met % (all) =
DIVIDE(
    CALCULATE(COUNTROWS(incident_facts), incident_facts[made_sla] = TRUE()),
    [Total Incidents]
)

SLA Met % (resolved) =
DIVIDE(
    CALCULATE(
        COUNTROWS(incident_facts),
        incident_facts[is_resolved] = TRUE(),
        incident_facts[made_sla] = TRUE()
    ),
    [Resolved Incidents]
)

Reopened % (resolved) =
DIVIDE(
    CALCULATE(
        COUNTROWS(incident_facts),
        incident_facts[is_resolved] = TRUE(),
        incident_facts[reopen_count] > 0
    ),
    [Resolved Incidents]
)

Reassigned % (resolved) =
DIVIDE(
    CALCULATE(
        COUNTROWS(incident_facts),
        incident_facts[is_resolved] = TRUE(),
        incident_facts[reassignment_count] > 0
    ),
    [Resolved Incidents]
)

Reassigned Share of Elapsed Hours =
DIVIDE(
    CALCULATE(
        SUM(incident_facts[resolution_hours]),
        incident_facts[reassignment_count] > 0
    ),
    SUM(incident_facts[resolution_hours])
)

Share of Resolved (by bucket) =
DIVIDE(
    [Resolved Incidents],
    CALCULATE([Resolved Incidents], ALL(incident_facts[resolution_bucket]))
)

---------------------------------------------------------
TABLE: state_durations
---------------------------------------------------------

Hours in State = SUM(state_durations[hours_in_state])

Avg Hours per Stretch = AVERAGE(state_durations[hours_in_state])

Share of All Time =
DIVIDE(
    [Hours in State],
    CALCULATE([Hours in State], ALL(state_durations[incident_state]))
)
